cmake_minimum_required(VERSION 3.16)
project(MiniDB VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(MINIDB_SOURCES src/main.cpp src/sql.cpp src/model.cpp src/storage.cpp src/engine.cpp src/bptree.cpp src/page_buffer.cpp src/lock_manager.cpp)
add_executable(minidb ${MINIDB_SOURCES})
target_include_directories(minidb PRIVATE include)
add_executable(minidb_unit tests/unit_cpp.cpp src/bptree.cpp src/page_buffer.cpp)
target_include_directories(minidb_unit PRIVATE include)
if(MSVC)
    target_compile_options(minidb PRIVATE /W4 /utf-8)
    target_compile_options(minidb_unit PRIVATE /UNDEBUG)
else()
    target_compile_options(minidb PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(minidb_unit PRIVATE -UNDEBUG)
    if(MINGW)
        set_target_properties(minidb PROPERTIES LINK_FLAGS "-static")
    endif()
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:minidb>)
    add_test(NAME concurrency COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/concurrency.py $<TARGET_FILE:minidb>)
endif()
add_test(NAME cpp_unit COMMAND minidb_unit)
